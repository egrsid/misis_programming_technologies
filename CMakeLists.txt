cmake_minimum_required(VERSION 3.25)
project(affine_practice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <mdspan>
int main() { double a[6]{}; std::mdspan<double, std::extents<int,2,3>> m(a); return m[0,0]; }" HAS_STD_MDSPAN)
if(NOT HAS_STD_MDSPAN)
  message(FATAL_ERROR "C++23 std::mdspan support is required")
endif()
add_library(affine_practice INTERFACE)
target_include_directories(affine_practice INTERFACE include)
add_executable(affine_tests tests/contracts.cpp)
target_link_libraries(affine_tests PRIVATE affine_practice)
enable_testing()
foreach(name storage identity add_scale transpose multiply layout translation scaling rotation composition)
  add_test(NAME ${name} COMMAND affine_tests ${name})
endforeach()
set_tests_properties(storage PROPERTIES LABELS scaffold)
set_tests_properties(identity add_scale transpose multiply layout PROPERTIES LABELS matrix)
set_tests_properties(translation scaling rotation composition PROPERTIES LABELS affine)
find_program(CLANG_FORMAT clang-format)
if(CLANG_FORMAT)
  file(GLOB_RECURSE FORMAT_FILES CONFIGURE_DEPENDS include/*.hpp tests/*.cpp)
  add_custom_target(format COMMAND ${CLANG_FORMAT} -i ${FORMAT_FILES})
  add_custom_target(format-check COMMAND ${CLANG_FORMAT} --dry-run --Werror ${FORMAT_FILES})
endif()
